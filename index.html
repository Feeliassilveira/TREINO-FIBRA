function handleBulkUpload(e) { const files = Array.from(e.target.files); if (!files.length) return; if (!state.images) state.images = {}; let fileIndex = 0; days.forEach(day => { workoutPlan[day].exercises.forEach(ex => { if (files[fileIndex]) { const reader = new FileReader(); reader.onload = function(ev) { const key = `${day}-${ex.id}`; state.images[key] = ev.target.result; localStorage.setItem("fc_images", JSON.stringify(state.images)); render(); }; reader.readAsDataURL(files[fileIndex]); fileIndex++; } }); }); }
